cmake_minimum_required(VERSION 3.16)
project(haetae_arm_guard)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(haetae_arm_guard SHARED src/controller.cpp)
target_compile_features(haetae_arm_guard PUBLIC cxx_std_17)
target_compile_options(haetae_arm_guard PRIVATE -Wall -Wextra -Werror)
target_include_directories(haetae_arm_guard PRIVATE include)
ament_target_dependencies(haetae_arm_guard controller_interface joint_trajectory_controller
  pluginlib rclcpp rclcpp_action std_msgs trajectory_msgs)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
install(TARGETS haetae_arm_guard LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(lease_test tests/lease_test.cpp)
  target_compile_features(lease_test PRIVATE cxx_std_17)
  target_compile_options(lease_test PRIVATE -Wall -Wextra -Werror)
  target_include_directories(lease_test PRIVATE include)
  add_test(NAME lease_policy COMMAND lease_test)
endif()
ament_package()
