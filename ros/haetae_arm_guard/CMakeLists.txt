cmake_minimum_required(VERSION 3.16)
project(haetae_arm_guard)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(diff_drive_controller REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(haetae_arm_guard SHARED src/controller.cpp src/base_controller.cpp)
target_compile_features(haetae_arm_guard PUBLIC cxx_std_17)
target_compile_options(haetae_arm_guard PRIVATE -Wall -Wextra -Werror)
target_include_directories(haetae_arm_guard PRIVATE include)
ament_target_dependencies(haetae_arm_guard controller_interface joint_trajectory_controller
  pluginlib rclcpp rclcpp_action std_msgs trajectory_msgs diff_drive_controller geometry_msgs)
target_link_libraries(haetae_arm_guard OpenSSL::Crypto)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
install(TARGETS haetae_arm_guard LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(lease_test tests/lease_test.cpp)
  target_compile_features(lease_test PRIVATE cxx_std_17)
  target_compile_options(lease_test PRIVATE -Wall -Wextra -Werror)
  target_include_directories(lease_test PRIVATE include)
  add_test(NAME lease_policy COMMAND lease_test)
  add_executable(permit_test tests/permit_test.cpp)
  target_compile_features(permit_test PRIVATE cxx_std_17)
  target_compile_options(permit_test PRIVATE -Wall -Wextra -Werror)
  target_include_directories(permit_test PRIVATE include)
  target_link_libraries(permit_test OpenSSL::Crypto)
  add_test(NAME controller_permit COMMAND permit_test)
  add_executable(payload_test tests/payload_test.cpp)
  target_compile_features(payload_test PRIVATE cxx_std_17)
  target_compile_options(payload_test PRIVATE -Wall -Wextra -Werror)
  target_include_directories(payload_test PRIVATE include)
  ament_target_dependencies(payload_test geometry_msgs trajectory_msgs)
  target_link_libraries(payload_test OpenSSL::Crypto)
  add_test(NAME exact_payload_vectors COMMAND payload_test)
endif()
ament_package()
