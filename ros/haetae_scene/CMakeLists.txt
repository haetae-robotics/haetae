cmake_minimum_required(VERSION 3.16)
project(haetae_scene)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(haetae_native_person SHARED src/native_person.cpp)
target_compile_features(haetae_native_person PUBLIC cxx_std_17)
target_compile_options(haetae_native_person PRIVATE -Wall -Wextra -Werror)
target_include_directories(haetae_native_person PRIVATE include)
target_link_libraries(haetae_native_person PRIVATE gz-sim8::gz-sim8
  gz-transport13::gz-transport13 gz-plugin2::register)
install(TARGETS haetae_native_person LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(motion_test tests/motion_test.cpp)
  target_compile_features(motion_test PRIVATE cxx_std_17)
  target_compile_options(motion_test PRIVATE -Wall -Wextra -Werror)
  target_include_directories(motion_test PRIVATE include)
  target_link_libraries(motion_test PRIVATE gz-sim8::gz-sim8)
  add_test(NAME native_person_motion COMMAND motion_test)
endif()
ament_package()
